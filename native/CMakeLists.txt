cmake_minimum_required(VERSION 3.16)
project(edgeshield_native VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(EDGESHIELD_BUILD_TESTS "Build native tests" ON)
find_package(Threads REQUIRED)
find_package(OpenSSL QUIET)
add_library(edgeshield_core src/detectors.cpp src/system_monitor.cpp src/risk.cpp src/sci.cpp src/http_server.cpp src/pcap_reader.cpp)
target_include_directories(edgeshield_core PUBLIC include)
target_link_libraries(edgeshield_core PUBLIC Threads::Threads)
if(OpenSSL_FOUND)
 target_link_libraries(edgeshield_core PUBLIC OpenSSL::Crypto)
 target_compile_definitions(edgeshield_core PRIVATE EDGESHIELD_WITH_OPENSSL=1)
 message(STATUS "OpenSSL found: SCI ledger uses SHA-256")
else()
 message(WARNING "OpenSSL not found: SCI ledger uses NON-CRYPTO development checksum; install libssl-dev for SHA-256")
endif()
target_compile_options(edgeshield_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(edgeshieldd src/main.cpp)
target_link_libraries(edgeshieldd PRIVATE edgeshield_core)
if(EDGESHIELD_BUILD_TESTS)
 enable_testing()
 add_executable(edgeshield_native_tests tests/test_core.cpp)
 target_link_libraries(edgeshield_native_tests PRIVATE edgeshield_core)
 add_test(NAME native_core COMMAND edgeshield_native_tests)
endif()
install(TARGETS edgeshield_core edgeshieldd)
